cmake_minimum_required(VERSION 3.5.0)

set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})

# set output directories for all configurations
foreach( OUTPUTCONFIG ${CMAKE_CONFIGURATION_TYPES} )
    string( TOUPPER ${OUTPUTCONFIG} OUTPUTCONFIG )
    set( CMAKE_LIBRARY_OUTPUT_DIRECTORY_${OUTPUTCONFIG} ${CMAKE_CURRENT_LIST_DIR}/bin/x64 )
    set( CMAKE_RUNTIME_OUTPUT_DIRECTORY_${OUTPUTCONFIG} ${CMAKE_CURRENT_LIST_DIR}/bin/x64 )
    set( CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${OUTPUTCONFIG} ${CMAKE_CURRENT_LIST_DIR}/bin/x64 )
endforeach( OUTPUTCONFIG CMAKE_CONFIGURATION_TYPES )

project(SSUSBLibProject VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_subdirectory(SSUSBLib)

add_custom_command(OUTPUT SSUSBLib COMMAND ${CMAKE_COMMAND} -E copy_if_different
    SSUSBLib/bin/x64/libusb-1.0.dll
    $<TARGET_FILE_DIR:SSUSBLib>
)

file(GLOB_RECURSE SSUSBLIB_TEST_SOURCES
    ${PROJECT_SOURCE_DIR}/SSUSBLibTest/*.cpp
    ${PROJECT_SOURCE_DIR}/SSUSBLibTest/*.h
)
add_executable(SSUSBLibTest ${SSUSBLIB_TEST_SOURCES})
target_include_directories(SSUSBLibTest PRIVATE ${PROJECT_SOURCE_DIR}/SSUSBLib)
target_link_libraries(SSUSBLibTest SSUSBLib)
